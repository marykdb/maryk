## Small model. Shared foundation.

Start with a model, store and query its values, then look back through its history. Each tab shows another part of the same data flow.

The same model also drives JSON, YAML, and ProtoBuf serialization. Stable field IDs let your data contract evolve over time.

[Set up your first model →](getting-started/)

<Tabs>
<TabItem label="Person.kt" icon="puzzle">

```kotlin
import maryk.core.models.RootDataModel
import maryk.core.properties.definitions.string

object Person : RootDataModel<Person>() {
    val firstName by string(index = 1u)
    val lastName by string(index = 2u)
}

val jane = Person.create {
    firstName with "Jane"
    lastName with "Doe"
}

Person.validate(jane)
val json = Person.Serializer.writeJson(jane)
```

</TabItem>
<TabItem label="Store.kt" icon="database">

```kotlin
import kotlinx.coroutines.runBlocking
import maryk.core.models.key
import maryk.core.query.requests.add
import maryk.core.query.requests.get
import maryk.datastore.memory.InMemoryDataStore

// Uses Person and jane from Person.kt.
fun storePerson() = runBlocking {
    InMemoryDataStore.open(
        dataModelsById = mapOf(1u to Person)
    ).use {
        val key = Person.key(jane)
        execute(Person.add(key to jane))
        val current = execute(Person.get(key))
        println(current.values.single().values)
    }
}
```

</TabItem>
<TabItem label="History.kt" icon="clock">

```kotlin
import kotlinx.coroutines.runBlocking
import maryk.core.models.key
import maryk.core.query.requests.add
import maryk.core.query.requests.get
import maryk.core.query.requests.getChanges
import maryk.datastore.memory.InMemoryDataStore

// Uses Person and jane from Person.kt.
fun readPersonHistory() = runBlocking {
    InMemoryDataStore.open(
        keepAllVersions = true,
        dataModelsById = mapOf(1u to Person)
    ).use {
        val key = Person.key(jane)
        execute(Person.add(key to jane))
        val changes = execute(Person.getChanges(key))
        val version = changes.changes.first()
            .changes.first().version
        val earlier = execute(
            Person.get(key, toVersion = version)
        )
        println(earlier.values.single().values)
    }
}
```
